cmake_minimum_required(VERSION 3.20)
project(course_kmeans LANGUAGES CXX)
add_executable(kmeans_cpu src/kmeans_cpu.cpp)
target_compile_features(kmeans_cpu PRIVATE cxx_std_17)
option(BUILD_CUDA "Build the historical POSIX CUDA version on Linux" OFF)
if(BUILD_CUDA)
    if(NOT UNIX)
        message(FATAL_ERROR "CUDA course code requires POSIX APIs; build under Linux or WSL")
    endif()
    enable_language(CUDA)
    add_executable(kmeans_cuda cuda/cuda_main.cu cuda/cuda_io.cu cuda/cuda_kmeans.cu cuda/cuda_wtime.cu)
    set_target_properties(kmeans_cuda PROPERTIES CUDA_STANDARD 14)
endif()
